<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Shopping Cart - Som Tam Thai</title>
    <script src="https://cdn.tailwindcss.com"></script>
    <script src="https://unpkg.com/react@18/umd/react.development.js" crossorigin></script>
    <script src="https://unpkg.com/react-dom@18/umd/react-dom.development.js" crossorigin></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
    <script src="https://unpkg.com/lucide@latest"></script>

    <style>
        /* Task 1-0.1: Design Tokens & Theme File */
        :root {
            --bg-cream: #FDFBF7;
            --primary-green: #1A4D2E;
            --accent-terracotta: #E36414;
            --highlight-gold: #FFB703;
        }
        @import url('https://fonts.googleapis.com/css2?family=DM+Sans:wght@400;500;700&family=Lora:ital,wght@0,400;0,600;1,400&display=swap');
        
        body { 
            background-color: var(--bg-cream); 
            font-family: 'DM Sans', sans-serif; 
            color: #333;
        }
        .display-font { 
            font-family: 'Lora', serif; 
        }
        .hide-scroll::-webkit-scrollbar {
            display: none;
        }
    </style>
    <script>
        tailwind.config = {
            theme: {
                extend: {
                    colors: {
                        brand: {
                            cream: '#FDFBF7',
                            green: '#1A4D2E',
                            terracotta: '#E36414',
                            gold: '#FFB703'
                        }
                    }
                }
            }
        }
    </script>
</head>
<body className="antialiased">
    <div id="root"></div>

    <script type="text/babel">
        const { ArrowLeft, Minus, Plus, X, Info, ShoppingBag, Receipt, AlertTriangle } = lucide;

        // Dummy Data
        const INITIAL_CART = [
            { 
                id: 'item-1', 
                productId: 1, 
                name: 'Authentic Som Tam', 
                price: 120, 
                quantity: 2, 
                variant: 'Medium Spice', 
                addons: ['Extra Roasted Peanuts (+MVR 15)'],
                image: '🥗',
                unitTotal: 135
            },
            { 
                id: 'item-2', 
                productId: 8, 
                name: 'Pad Thai', 
                price: 160, 
                quantity: 1, 
                variant: 'Mild', 
                addons: [],
                image: '🍜',
                unitTotal: 160
            }
        ];

        // Wiremock toggle for open/closed state demonstration
        const OPERATING_HOURS = { isOpen: false }; 

        // Reused NavBar Component
        const NavBar = ({ cartCount }) => (
            <nav className="sticky top-0 z-50 bg-white shadow-sm px-4 py-3 flex justify-between items-center border-b border-gray-100">
                <div className="flex items-center gap-2">
                    <span className="display-font font-bold text-xl text-brand-green">Som Tam Thai</span>
                </div>
                <div className="hidden md:flex gap-6 font-medium text-sm">
                    <a href="index.html" className="text-gray-500 hover:text-brand-terracotta transition-colors">Menu</a>
                    <a href="reservations.html" className="hover:text-brand-terracotta transition-colors">Reservations</a>
                    <a href="orders.html" className="hover:text-brand-terracotta transition-colors">Orders</a>
                    <a href="account.html" className="hover:text-brand-terracotta transition-colors">Account</a>
                </div>
                <div className="flex items-center gap-4">
                    <a href="cart.html" className="relative p-2 text-brand-green">
                        <i data-lucide="shopping-bag" className="w-6 h-6 text-brand-terracotta"></i>
                        {cartCount > 0 && (
                            <span className="absolute top-0 right-0 inline-flex items-center justify-center px-2 py-1 text-xs font-bold leading-none text-white transform translate-x-1/4 -translate-y-1/4 bg-brand-terracotta rounded-full">
                                {cartCount}
                            </span>
                        )}
                    </a>
                </div>
            </nav>
        );

        // Task 1-3.1: Cart Line Item Component
        const CartItem = ({ item, updateQty, removeItem }) => (
            <div className="flex gap-4 py-5 border-b border-gray-100 last:border-0 relative bg-white sm:bg-transparent sm:p-0 sm:py-5 sm:border-b">
                <div className="w-20 h-20 bg-gray-100 rounded-lg flex items-center justify-center text-3xl flex-shrink-0">
                    {item.image}
                </div>
                
                <div className="flex-1 flex flex-col">
                    <div className="flex justify-between items-start mb-1">
                        <h3 className="font-bold text-gray-900 leading-tight pr-8">{item.name}</h3>
                        <button 
                            onClick={() => removeItem(item.id)}
                            className="text-gray-400 hover:text-red-500 transition-colors absolute top-5 right-0 sm:static sm:p-1"
                            aria-label="Remove item"
                        >
                            <i data-lucide="x" className="w-5 h-5"></i>
                        </button>
                    </div>
                    
                    <div className="text-sm text-gray-500 mb-3">
                        <p>{item.variant}</p>
                        {item.addons.map((addon, idx) => (
                            <p key={idx} className="text-xs">{addon}</p>
                        ))}
                    </div>
                    
                    <div className="mt-auto flex items-center justify-between">
                        <div className="flex items-center border border-gray-300 rounded-md overflow-hidden h-8 bg-white">
                            <button 
                                onClick={() => updateQty(item.id, item.quantity - 1)}
                                className="w-8 h-full flex items-center justify-center text-gray-500 hover:bg-gray-100 transition-colors"
                            >
                                <i data-lucide="minus" className="w-3 h-3"></i>
                            </button>
                            <div className="w-8 h-full flex items-center justify-center font-semibold text-sm border-x border-gray-300">
                                {item.quantity}
                            </div>
                            <button 
                                onClick={() => updateQty(item.id, item.quantity + 1)}
                                className="w-8 h-full flex items-center justify-center text-brand-green hover:bg-gray-100 transition-colors"
                            >
                                <i data-lucide="plus" className="w-3 h-3"></i>
                            </button>
                        </div>
                        
                        <div className="font-bold text-brand-green text-right">
                            MVR {item.unitTotal * item.quantity}
                        </div>
                    </div>
                </div>
            </div>
        );

        // Main Cart App Component
        const App = () => {
            const [items, setItems] = React.useState(INITIAL_CART);
            const [notes, setNotes] = React.useState("");

            React.useEffect(() => {
                lucide.createIcons();
            });

            // Cart Actions
            const updateQty = (id, newQty) => {
                if (newQty < 1) return;
                setItems(items.map(item => item.id === id ? { ...item, quantity: newQty } : item));
            };

            const removeItem = (id) => {
                setItems(items.filter(item => item.id !== id));
            };

            // Totals Calculation
            const subtotal = items.reduce((sum, item) => sum + (item.unitTotal * item.quantity), 0);
            const deliveryFee = subtotal > 0 ? 35 : 0; // Flat fee dummy logic
            const gst = subtotal * 0.08; // 8% GST
            const grandTotal = subtotal + deliveryFee + gst;
            const totalItemsCount = items.reduce((sum, item) => sum + item.quantity, 0);

            // Empty State Component
            if (items.length === 0) {
                return (
                    <div className="min-h-screen flex flex-col bg-brand-cream">
                        <NavBar cartCount={0} />
                        <div className="flex-grow flex flex-col items-center justify-center p-6 text-center">
                            <div className="w-32 h-32 bg-gray-100 rounded-full flex items-center justify-center mb-6 shadow-inner">
                                <span className="text-6xl">🥣</span>
                            </div>
                            <h2 className="display-font text-2xl font-bold text-brand-green mb-2">Your cart is empty</h2>
                            <p className="text-gray-500 mb-8 max-w-sm">Looks like you haven't added any authentic Thai dishes to your order yet.</p>
                            <a href="index.html" className="bg-brand-terracotta hover:bg-orange-700 text-white font-bold py-3 px-8 rounded-lg shadow-sm transition-colors">
                                Browse Menu
                            </a>
                        </div>
                    </div>
                );
            }

            // Populated Cart View
            return (
                <div className="min-h-screen flex flex-col bg-brand-cream pb-24 lg:pb-12">
                    <NavBar cartCount={totalItemsCount} />

                    <main className="flex-grow max-w-7xl mx-auto w-full px-4 py-6 md:py-10">
                        <div className="flex items-center justify-between mb-6">
                            <h1 className="display-font text-3xl font-bold text-gray-900">Your Cart</h1>
                            <span className="text-gray-500 text-sm font-medium">{totalItemsCount} {totalItemsCount === 1 ? 'item' : 'items'}</span>
                        </div>

                        {/* Preorder Alert Banner */}
                        {!OPERATING_HOURS.isOpen && (
                            <div className="bg-amber-50 border border-amber-200 rounded-lg p-4 mb-8 flex items-start shadow-sm">
                                <i data-lucide="info" className="w-5 h-5 text-amber-500 mr-3 mt-0.5 flex-shrink-0"></i>
                                <div>
                                    <h3 className="text-amber-800 font-bold text-sm mb-1">We're currently closed</h3>
                                    <p className="text-amber-700 text-sm">You are placing a preorder. You will select your desired delivery or pickup time during checkout.</p>
                                </div>
                            </div>
                        )}

                        {/* Two-Column Layout */}
                        <div className="flex flex-col lg:flex-row gap-8 lg:gap-12">
                            
                            {/* Left Column: Items & Notes */}
                            <div className="flex-1">
                                <div className="bg-white rounded-xl shadow-sm border border-gray-100 p-4 sm:p-6 mb-6">
                                    <div className="flex flex-col gap-2">
                                        {items.map(item => (
                                            <CartItem 
                                                key={item.id} 
                                                item={item} 
                                                updateQty={updateQty} 
                                                removeItem={removeItem} 
                                            />
                                        ))}
                                    </div>
                                </div>

                                <div className="bg-white rounded-xl shadow-sm border border-gray-100 p-4 sm:p-6">
                                    <h3 className="font-bold text-gray-900 mb-3 flex items-center">
                                        <i data-lucide="receipt" className="w-4 h-4 mr-2 text-gray-400"></i>
                                        Order Notes
                                    </h3>
                                    <textarea 
                                        rows="3" 
                                        placeholder="Any allergy requests or special delivery instructions? (e.g. Leave at reception, make it extra spicy)"
                                        value={notes}
                                        onChange={(e) => setNotes(e.target.value)}
                                        className="w-full border border-gray-300 rounded-lg p-3 text-sm focus:border-brand-green focus:ring-1 focus:ring-brand-green focus:outline-none"
                                    ></textarea>
                                </div>
                            </div>

                            {/* Right Column: Sticky Summary Panel */}
                            <div className="w-full lg:w-96">
                                <div className="bg-white rounded-xl shadow-sm border border-gray-100 p-6 lg:sticky lg:top-24">
                                    <h2 className="font-bold text-lg text-gray-900 mb-6 border-b border-gray-100 pb-4">Order Summary</h2>
                                    
                                    <div className="space-y-4 text-sm mb-6">
                                        <div className="flex justify-between text-gray-600">
                                            <span>Subtotal</span>
                                            <span className="font-medium text-gray-900">MVR {subtotal.toFixed(2)}</span>
                                        </div>
                                        <div className="flex justify-between text-gray-600">
                                            <span>Delivery Fee <span className="text-xs text-gray-400 ml-1">(Malé)</span></span>
                                            <span className="font-medium text-gray-900">MVR {deliveryFee.toFixed(2)}</span>
                                        </div>
                                        <div className="flex justify-between text-gray-600">
                                            <span>GST (8%)</span>
                                            <span className="font-medium text-gray-900">MVR {gst.toFixed(2)}</span>
                                        </div>
                                    </div>

                                    <div className="border-t border-gray-100 pt-4 mb-8">
                                        <div className="flex justify-between items-center">
                                            <span className="font-bold text-lg text-gray-900">Grand Total</span>
                                            <span className="font-bold text-2xl text-brand-green">MVR {grandTotal.toFixed(2)}</span>
                                        </div>
                                    </div>

                                    <a 
                                        href="checkout.html" 
                                        className="w-full bg-brand-green hover:bg-green-900 text-white font-bold py-4 px-4 rounded-lg shadow-sm transition-colors flex justify-center items-center text-lg"
                                    >
                                        Proceed to Checkout
                                    </a>
                                    
                                    <div className="mt-4 text-center">
                                        <a href="index.html" className="text-sm text-gray-500 hover:text-brand-green font-medium transition-colors">
                                            Continue Shopping
                                        </a>
                                    </div>
                                </div>
                            </div>

                        </div>
                    </main>
                </div>
            );
        };

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
</body>
</html>
